DML generated using XML to DML utility:

    XML Example 1-> 

XML format:
<Order id="101">
  <Customer>John Doe</Customer>
  <Items>
    <Product>Laptop</Product>
    <Product>Mouse</Product>
  </Items>
</Order>

    DML generated:
utf8 record
  /* Mapping for the 'id' attribute */
  decimal("\0") id;
 
  /* Mapping for the 'Customer' element */
  string("\0") Customer;
 
  /* Mapping for the 'Items' wrapper and nested 'Product' elements */
  record
    string("\0")[int] Product; // [int] denotes a variable-length vector
  } Items;
 
  string("\0") newline = "\n";
end

DML generated:
/* Generated by xml-to-dml utility */

type policy_address_t = record
  string("\0") Street @style="element";
  string("\0") City @style="element";
  decimal("\0") Zip @style="element";
end;

type vehicle_info_t = record
  string("\0") vehicle_type @style="attribute", name="type";
  string("\0") Make @style="element";
  decimal("\0") Year @style="element";
  decimal(".2") Premium @style="element"; // Interpreted as a decimal with 2 scale
end;

metadata type = record
  // The root attribute
  string("\0") id @style="attribute";

  // Nested record for Insured details
  record
    string("\0") Name @style="element";
    policy_address_t Address;
  } Insured @style="element";

  // Vector of records for Vehicles (handling maxOccurs="unbounded")
  record
    vehicle_info_t[int] Vehicle;
  } Vehicles @style="element";

  string("\0") newline = "\n";
end;

DML generated using Cobol to DML utility

COBOL code:

01  EMPLOYEE-RECORD.
    05  EMP-ID              PIC 9(05).
    05  EMP-NAME            PIC X(20).
    05  EMP-DETAILS.
        10  EMP-DEPT        PIC X(10).
        10  EMP-SALARY      PIC S9(7)V99  COMP-3.
    05  PROJECT-COUNT       PIC 9(02).
    05  PROJECTS            OCCURS 1 TO 5 TIMES
                            DEPENDING ON PROJECT-COUNT.
        10  PROJ-ID         PIC X(04).

DML generated:

/* Generated from COBOL copybook */

record
  ascii decimal(5) emp_id;                 /* PIC 9(05) */
  string(20) emp_name;                     /* PIC X(20) */
 
  record
    string(10) emp_dept;                   /* PIC X(10) */
    packed decimal(9,2) emp_salary;        /* PIC S9(7)V99 COMP-3 */
  } emp_details;
 
  ascii decimal(2) project_count;          /* PIC 9(02) */
 
  /* OCCURS DEPENDING ON clause becomes a variable length vector */
  record
    string(4) proj_id;                     /* PIC X(04) */
  } projects[project_count];              
 
  string(1) newline = "\n";
end

Hive DML

Table structure:

CREATE TABLE sales_data (
    transaction_id  BIGINT,
    customer_name   STRING,
    amount          DECIMAL(10,2),
    items_list      ARRAY<STRING>,
    meta_info       MAP<STRING, STRING>
)
PARTITIONED BY (year INT, month INT)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY '\001';

DML generated:

/* Generated from Hive Table: sales_data */

record
  /* Basic Hive Types */
  integer(8) transaction_id;                  /* BIGINT */
  string("\001") customer_name;               /* STRING */
  decimal(10,2) amount;                       /* DECIMAL(10,2) */

  /* Hive ARRAY becomes a DML Vector */
  record
    string("\001")[int] item;
  } items_list;

  /* Hive MAP becomes a Vector of Key-Value pairs */
  record
    record
      string("\001") key;
      string("\001") value;
    } [int] map_entries;
  } meta_info;

  /* Partition columns are often included as standard fields */
  integer(4) year;                            /* INT */
  integer(4) month;                           /* INT */

  string("\n") newline;
end
